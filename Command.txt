Coupons API - Assessment Part 1 - Commands
===========================================
All commands are run in PowerShell from the project folder (coupons-api).
LocalStack should be running locally on http://localhost:4566.


0. SETUP
--------
# Check LocalStack is running
lstk status

# Install AWS CLI + awslocal wrapper
py -m pip install awscli awscli-local

# Install project dependencies (Jest for unit tests, Newman for API tests)
npm install

# Dummy credentials for LocalStack
$env:AWS_ACCESS_KEY_ID = "test"
$env:AWS_SECRET_ACCESS_KEY = "test"
$env:AWS_DEFAULT_REGION = "us-east-1"


1. CREATE IAM ROLE coupons_lambda_role
--------------------------------------
awslocal iam create-role `
  --role-name coupons_lambda_role `
  --assume-role-policy-document file://iam/trust-policy.json


2. CLOUDWATCH LOGS PERMISSIONS (only log groups prefixed /aws/lambda/coupons*)
------------------------------------------------------------------------------
awslocal iam create-policy `
  --policy-name coupons_lambda_logs_policy `
  --policy-document file://iam/coupons-logs-policy.json

awslocal iam attach-role-policy `
  --role-name coupons_lambda_role `
  --policy-arn "arn:aws:iam::000000000000:policy/coupons_lambda_logs_policy"

# Verify
awslocal iam list-attached-role-policies --role-name coupons_lambda_role
awslocal iam get-policy-version `
  --policy-arn "arn:aws:iam::000000000000:policy/coupons_lambda_logs_policy" `
  --version-id v1


3. CREATE LAMBDA FUNCTION coupons_list
--------------------------------------
Compress-Archive -Path src\index.js -DestinationPath coupons_list.zip -Force

awslocal lambda create-function `
  --function-name coupons_list `
  --runtime nodejs20.x `
  --handler index.handler `
  --zip-file fileb://coupons_list.zip `
  --role "arn:aws:iam::000000000000:role/coupons_lambda_role"

awslocal lambda wait function-active-v2 --function-name coupons_list


4. INVOKE LAMBDA WITH A SAMPLE API GATEWAY EVENT (writes output.json)
---------------------------------------------------------------------
awslocal lambda invoke `
  --function-name coupons_list `
  --payload file://events/coupons_list_event.json `
  output.json

Get-Content output.json


5. ADD GET /coupons_poc ENDPOINT TO API GATEWAY
-----------------------------------------------
# Create the REST API (custom id "coupons" keeps the URL stable)
awslocal apigateway create-rest-api `
  --name coupons-api `
  --tags _custom_id_=coupons

# Get the root "/" resource id
$ROOT_ID = awslocal apigateway get-resources `
  --rest-api-id coupons `
  --query "items[?path=='/'].id" `
  --output text

# Create the /coupons_poc resource
$RESOURCE_ID = awslocal apigateway create-resource `
  --rest-api-id coupons `
  --parent-id $ROOT_ID `
  --path-part coupons_poc `
  --query "id" `
  --output text

# Add the GET method
awslocal apigateway put-method `
  --rest-api-id coupons `
  --resource-id $RESOURCE_ID `
  --http-method GET `
  --authorization-type NONE

# Connect GET to the coupons_list Lambda (proxy integration)
awslocal apigateway put-integration `
  --rest-api-id coupons `
  --resource-id $RESOURCE_ID `
  --http-method GET `
  --type AWS_PROXY `
  --integration-http-method POST `
  --uri "arn:aws:apigateway:us-east-1:lambda:path/2015-03-31/functions/arn:aws:lambda:us-east-1:000000000000:function:coupons_list/invocations"

# Allow API Gateway to invoke the Lambda
awslocal lambda add-permission `
  --function-name coupons_list `
  --statement-id apigateway-coupons-poc `
  --action lambda:InvokeFunction `
  --principal apigateway.amazonaws.com `
  --source-arn "arn:aws:execute-api:us-east-1:000000000000:coupons/*/GET/coupons_poc"

# Deploy the API to the "dev" stage
awslocal apigateway create-deployment `
  --rest-api-id coupons `
  --stage-name dev

# Test the endpoint
curl.exe http://localhost:4566/restapis/coupons/dev/_user_request_/coupons_poc
