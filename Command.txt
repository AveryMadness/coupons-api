DYNAMODB CAPACITY UNIT CALCULATIONS (Assessment Part 2 Requirement 1):
----------------------------------------------------------------------
• Read Requirements: 50 eventually consistent reads per second, 1 KB item size.
  - 1 Read Capacity Unit (RCU) = 1 strongly consistent read of up to 4 KB/sec
    OR 2 eventually consistent reads of up to 4 KB/sec.
  - 1 KB item size rounds up to 4 KB.
  - Since reads are eventually consistent: 50 * (1/2) = 25 RCU.
• Write Requirements: 10 writes per second, 1 KB item size.
  - 1 Write Capacity Unit (WCU) = 1 write of up to 1 KB/sec.
  - Since item size is 1 KB: 10 * 1 = 10 WCU.
• Conclusion: ProvisionedThroughput is ReadCapacityUnits=25, WriteCapacityUnits=10.

2. CREATE DYNAMODB TABLE "coupons"
-----------------------------------
# Create the table with 25 RCU and 10 WCU
awslocal dynamodb create-table `
  --cli-input-json file://dynamodb/create-table.json

# Wait for the table to become active
awslocal dynamodb wait table-exists --table-name coupons

# Verify table creation and provisioned capacity
awslocal dynamodb describe-table --table-name coupons


3. UPDATE DYNAMODB TABLE WITH SECONDARY INDEX (provider_id & campaign_id)
--------------------------------------------------------------------------
# Add Global Secondary Index for performant queries on provider_id and campaign_id
awslocal dynamodb update-table `
  --table-name coupons `
  --attribute-definitions AttributeName=provider_id,AttributeType=S AttributeName=campaign_id,AttributeType=S `
  --global-secondary-index-updates file://dynamodb/gsi-update.json

# Verify the GSI was registered
awslocal dynamodb describe-table --table-name coupons --query "Table.GlobalSecondaryIndexes"

6. CONFIGURE API GATEWAY
------------------------
# Create REST API
awslocal apigateway create-rest-api `
  --name coupons-api `
  --tags _custom_id_=coupons

# Get root resource ID ("/")
$ROOT_ID = awslocal apigateway get-resources `
  --rest-api-id coupons `
  --query "items[?path=='/'].id" `
  --output text

# 6.1 /coupons_poc resource and GET method
$POC_ID = awslocal apigateway create-resource `
  --rest-api-id coupons `
  --parent-id $ROOT_ID `
  --path-part coupons_poc `
  --query "id" `
  --output text

awslocal apigateway put-method `
  --rest-api-id coupons `
  --resource-id $POC_ID `
  --http-method GET `
  --authorization-type NONE

awslocal apigateway put-integration `
  --rest-api-id coupons `
  --resource-id $POC_ID `
  --http-method GET `
  --type AWS_PROXY `
  --integration-http-method POST `
  --uri "arn:aws:apigateway:us-east-1:lambda:path/2015-03-31/functions/arn:aws:lambda:us-east-1:000000000000:function:coupons_list/invocations"

awslocal lambda add-permission `
  --function-name coupons_list `
  --statement-id apigateway-coupons-poc `
  --action lambda:InvokeFunction `
  --principal apigateway.amazonaws.com `
  --source-arn "arn:aws:execute-api:us-east-1:000000000000:coupons/*/GET/coupons_poc"

# 6.2 /coupons resource
$COUPONS_ID = awslocal apigateway create-resource `
  --rest-api-id coupons `
  --parent-id $ROOT_ID `
  --path-part coupons `
  --query "id" `
  --output text

# 6.3 /coupons/{id} resource
$COUPON_PARAM_ID = awslocal apigateway create-resource `
  --rest-api-id coupons `
  --parent-id $COUPONS_ID `
  --path-part '{id}' `
  --query "id" `
  --output text

# GET /coupons/{id} -> coupons_get_by_id
awslocal apigateway put-method `
  --rest-api-id coupons `
  --resource-id $COUPON_PARAM_ID `
  --http-method GET `
  --authorization-type NONE

awslocal apigateway put-integration `
  --rest-api-id coupons `
  --resource-id $COUPON_PARAM_ID `
  --http-method GET `
  --type AWS_PROXY `
  --integration-http-method POST `
  --uri "arn:aws:apigateway:us-east-1:lambda:path/2015-03-31/functions/arn:aws:lambda:us-east-1:000000000000:function:coupons_get_by_id/invocations"

awslocal lambda add-permission `
  --function-name coupons_get_by_id `
  --statement-id apigateway-coupons-get-by-id `
  --action lambda:InvokeFunction `
  --principal apigateway.amazonaws.com `
  --source-arn "arn:aws:execute-api:us-east-1:000000000000:coupons/*/GET/coupons/*"

# PUT /coupons/{id} -> coupons_update
awslocal apigateway put-method `
  --rest-api-id coupons `
  --resource-id $COUPON_PARAM_ID `
  --http-method PUT `
  --authorization-type NONE

awslocal apigateway put-integration `
  --rest-api-id coupons `
  --resource-id $COUPON_PARAM_ID `
  --http-method PUT `
  --type AWS_PROXY `
  --integration-http-method POST `
  --uri "arn:aws:apigateway:us-east-1:lambda:path/2015-03-31/functions/arn:aws:lambda:us-east-1:000000000000:function:coupons_update/invocations"

awslocal lambda add-permission `
  --function-name coupons_update `
  --statement-id apigateway-coupons-update `
  --action lambda:InvokeFunction `
  --principal apigateway.amazonaws.com `
  --source-arn "arn:aws:execute-api:us-east-1:000000000000:coupons/*/PUT/coupons/*"

# Deploy API to "dev" stage
awslocal apigateway create-deployment `
  --rest-api-id coupons `
